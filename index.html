// Thay đổi hàm initApp ban đầu để mặc định bật Màn hình Đăng nhập async function initApp() { try { await initDatabase(); await seedInitialDataIfNeeded(); await loadClassesList(); // Luôn hiển thị màn hình khóa/đăng nhập ban đầu khi mở link document.getElementById('initialAuthScreen').classList.remove('hidden'); } catch (err) { console.error("Init error:", err); } } // Chuyển bước chọn vai trò function showLoginRole(role) { if (role === 'student') { document.getElementById('initialAuthScreen').classList.add('hidden'); forceSwitchMode('student'); } else if (role === 'teacher') { document.getElementById('authRoleSelection').classList.add('hidden'); document.getElementById('authTeacherPasswordForm').classList.remove('hidden'); document.getElementById('mainTeacherPasswordInput').focus(); } } // Quay lại chọn vai trò function backToRoleSelection() { document.getElementById('authTeacherPasswordForm').classList.add('hidden'); document.getElementById('authRoleSelection').classList.remove('hidden'); document.getElementById('mainTeacherPasswordInput').value = ""; } // Xác thực mật khẩu đăng nhập Giáo viên ban đầu function mainAuthenticateTeacher() { const inputPass = document.getElementById('mainTeacherPasswordInput').value.trim(); if (inputPass === teacherPassword) { document.getElementById('initialAuthScreen').classList.add('hidden'); backToRoleSelection(); // Reset form cho lần sau forceSwitchMode('teacher'); showToast("Đăng nhập Giáo viên thành công!", "success"); } else { showToast("Mật khẩu Giáo viên không chính xác!", "error"); } }
